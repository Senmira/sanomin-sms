   105→function toDateInput(iso?: string | null): string {
   106→  if (!iso) return ''
   107→  const d = new Date(iso)
   108→  if (isNaN(d.getTime())) return ''
   109→  return d.toISOString().slice(0, 10)
   110→}
   111→
   112→function csvEscape(v: unknown): string {
   113→  const s = String(v ?? '')
   114→  if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   115→  return s
   116→}

   118→function exportTeachersCsv(rows: TeacherRow[]): void {
   119→  const headers = [
   120→    'Teacher ID',
   121→    'Fingerprint ID',
   122→    'Full Name',
   123→    'Type',
   124→    'Gender',
   125→    'Phone',
   126→    'Email',
   127→    'NIC',
   128→    'Qualification',
   129→    'Specialization',
   130→    'Status',
   131→    'Hire Date',
   132→    'Monthly Rate (LKR)',
   133→    'Classes',
   134→  ]
   135→  const lines = rows.map((r) =>
   136→    [
   137→      r.teacherId,
   138→      r.fingerprintId || '',
   139→      r.fullName,
   140→      r.type,
   141→      r.gender || '',
   142→      r.phone || '',
   143→      r.email || '',
   144→      r.nic || '',
   145→      r.qualification || '',
   146→      r.specialization || '',
   147→      r.status,
   148→      r.hireDate ? toDateInput(r.hireDate) : '',
   149→      String(r.monthlyRate ?? 0),
   150→      String(r._count?.classes ?? 0),
   151→    ]
   152→      .map(csvEscape)
   153→      .join(','),
   154→  )
   155→  const csv = [headers.join(','), ...lines].join('\n')
   156→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   157→  const url = URL.createObjectURL(blob)
   158→  const a = document.createElement('a')
   159→  a.href = url
   160→  a.download = `sanomin-teachers-${new Date().toISOString().slice(0, 10)}.csv`
   161→  document.body.appendChild(a)
   162→  a.click()
   163→  document.body.removeChild(a)
   164→  URL.revokeObjectURL(url)
   165→}

  1178→// ─── Manage Fingerprint dialog (simulated scanner) ─────────────────────────
  1179→function ManageFingerprintDialog({
  1180→  teacher,
  1181→  onClose,
  1182→  onSaved,
  1183→}: {
  1184→  teacher: TeacherRow
  1185→  onClose: () => void
  1186→  onSaved: (msg: string) => void
  1187→}) {
  1188→  const [scanning, setScanning] = useState(false)
  1189→  const [stage, setStage] = useState<'idle' | 'scanning' | 'enrolled'>('idle')
  1190→  const [newFp, setNewFp] = useState<string | null>(null)
  1191→  const [saving, setSaving] = useState(false)
  1192→  const [err, setErr] = useState<string | null>(null)
  1193→
  1194→  const currentFp = teacher.fingerprintId
  1195→
  1196→  const runScan = () => {
  1197→    if (scanning) return
  1198→    setScanning(true)
  1199→    setStage('scanning')
  1200→    setErr(null)
  1201→    // Simulate the fingerprint reader taking ~1.8s to capture the template
  1202→    setTimeout(() => {
  1203→      const generated = randomFingerprintId()
  1204→      setNewFp(generated)
  1205→      setStage('enrolled')
  1206→      setScanning(false)
  1207→    }, 1800)
  1208→  }
  1209→
  1210→  const commit = async (fpToSet: string | null) => {
  1211→    setSaving(true)
  1212→    setErr(null)
  1213→    try {
  1214→      await api<TeacherRow>(`/api/teachers/${teacher.id}`, {
  1215→        method: 'PUT',
  1216→        body: JSON.stringify({ fingerprintId: fpToSet }),
  1217→      })
  1218→      onSaved(
  1219→        fpToSet
  1220→          ? `Fingerprint ${fpToSet} enrolled for ${teacher.fullName}`
  1221→          : `Fingerprint removed from ${teacher.fullName}`,
  1222→      )
  1223→    } catch (e) {
  1224→      const msg = e instanceof Error ? e.message : 'Failed to save fingerprint'
  1225→      setErr(msg)
  1226→      toast.error(msg)
  1227→    } finally {
  1228→      setSaving(false)
  1229→    }
  1230→  }
  1231→
  1232→  const handleEnroll = () => {
  1233→    if (newFp) commit(newFp)
  1234→  }
