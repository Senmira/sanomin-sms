    93→    93→function toDateInput(iso?: string | null): string {
    94→    94→  if (!iso) return ''
    95→    95→  const d = new Date(iso)
    96→    96→  if (isNaN(d.getTime())) return ''
    97→    97→  const pad = (n: number) => String(n).padStart(2, '0')
    98→    98→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
    99→    99→}
   100→   100→
   101→   101→function toDateTimeInput(iso?: string | null): string {
   102→   102→  if (!iso) return ''
   103→   103→  const d = new Date(iso)
   104→   104→  if (isNaN(d.getTime())) return ''
   105→   105→  const pad = (n: number) => String(n).padStart(2, '0')
   106→   106→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   107→   107→}

   109→   109→function todayStr(): string {
   110→   110→  const now = new Date()
   111→   111→  const pad = (n: number) => String(n).padStart(2, '0')
   112→   112→  return `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`
   113→   113→}

   177→   177→function ScannerPanel({ onScanSuccess, onOpenManual }: ScannerPanelProps) {
   178→   178→  const [tab, setTab] = useState<'barcode' | 'fingerprint'>('barcode')
   179→   179→  const [barcodeValue, setBarcodeValue] = useState('')
   180→   180→  const [fpValue, setFpValue] = useState('')
   181→   181→  const [scanning, setScanning] = useState(false)
   182→   182→  const [result, setResult] = useState<ScanResponse | null>(null)
   183→   183→  const [error, setError] = useState<string | null>(null)
   184→   184→  const [flashTick, setFlashTick] = useState(0)
   185→   185→  const [teachers, setTeachers] = useState<
   186→   186→    Array<{ id: string; teacherId: string; fingerprintId: string | null; fullName: string }>
   187→   187→  >([])
   188→   188→  const [teachersLoading, setTeachersLoading] = useState(false)
   189→   189→  const barcodeInputRef = useRef<HTMLInputElement>(null)
   190→   190→  const fpInputRef = useRef<HTMLInputElement>(null)
   191→   191→
   192→   192→  // Lazy-load teachers when the fingerprint tab opens (for the random demo button).
   193→   193→  useEffect(() => {
   194→   194→    if (tab !== 'fingerprint' || teachers.length > 0) return
   195→   195→    let alive = true
   196→   196→    setTeachersLoading(true)
   197→   197→    api<{ data: Array<{ id: string; teacherId: string; fingerprintId: string | null; fullName: string }> }>(
   198→   198→      '/api/teachers?limit=100',
   199→   199→    )
   200→   200→      .then((res) => {
   201→   201→        if (!alive) return
   202→   202→        const list = (res.data || []).filter((t) => !!t.fingerprintId)
   203→   203→        setTeachers(list)
   204→   204→      })
   205→   205→      .catch(() => {
   206→   206→        /* non-fatal — demo button will just disable itself */
   207→   207→      })
   208→   208→      .finally(() => {
   209→   209→        if (alive) setTeachersLoading(false)
   210→   210→      })
   211→   211→    return () => {
   212→   212→      alive = false
   213→   213→    }
   214→   214→  }, [tab, teachers.length])
   215→   215→
   216→   216→  // Refocus the active input whenever the tab changes.
   217→   217→  useEffect(() => {
   218→   218→    if (tab === 'barcode') barcodeInputRef.current?.focus()
   219→   219→    else fpInputRef.current?.focus()
   220→   220→  }, [tab])
   221→   221→
   222→   222→  const runScan = useCallback(
   223→   223→    async (method: 'barcode' | 'fingerprint' | 'manual', value: string) => {
   224→   224→      if (!value.trim()) return
   225→   225→      setScanning(true)
   226→   226→      setError(null)
   227→   227→      try {
   228→   228→        const res = await api<ScanResponse>('/api/attendance/scan', {
   229→   229→          method: 'POST',
   230→   230→          body: JSON.stringify({ method, value: value.trim() }),
   231→   231→        })
   232→   232→        setResult(res)
   233→   233→        setFlashTick((t) => t + 1)
   234→   234→        onScanSuccess()
   235→   235→        const verb = res.action === 'check-in' ? 'Checked in' : 'Checked out'
   236→   236→        toast.success(`${verb}: ${res.person.name}`, {
   237→   237→          description: `${res.person.type} · ${res.person.ref}`,
   238→   238→        })
   239→   239→      } catch (err) {
   240→   240→        const msg = err instanceof Error ? err.message : 'Scan failed'
   241→   241→        setError(msg)
   242→   242→        toast.error(msg)
   243→   243→      } finally {
   244→   244→        setScanning(false)
   245→   245→      }
   246→   246→    },
   247→   247→    [onScanSuccess],
   248→   248→  )
   249→   249→
   250→   250→  const handleBarcodeKey = (e: React.KeyboardEvent<HTMLInputElement>) => {
   251→   251→    if (e.key === 'Enter') {
   252→   252→      e.preventDefault()
   253→   253→      const v = barcodeValue
   254→   254→      setBarcodeValue('')
   255→   255→      runScan('barcode', v)
   256→   256→    }
   257→   257→  }
   258→   258→
   259→   259→  const handleFingerprintKey = (e: React.KeyboardEvent<HTMLInputElement>) => {
   260→   260→    if (e.key === 'Enter') {
   261→   261→      e.preventDefault()
   262→   262→      const v = fpValue
   263→   263→      setFpValue('')
   264→   264→      runScan('fingerprint', v)
   265→   265→    }
   266→   266→  }
   267→   267→
   268→   268→  const handleRandomFingerprint = () => {
   269→   269→    const enrolled = teachers.filter((t) => !!t.fingerprintId)
   270→   270→    if (enrolled.length === 0) {
   271→   271→      toast.error('No teachers with enrolled fingerprints')
   272→   272→      return
   273→   273→    }
   274→   274→    const pick = enrolled[Math.floor(Math.random() * enrolled.length)]
   275→   275→    setFpValue('')
   276→   276→    runScan('fingerprint', pick.fingerprintId as string)
   277→   277→  }
   278→   278→
   279→   279→  const reset = () => {
   280→   280→    setResult(null)
   281→   281→    setError(null)
   282→   282→    setBarcodeValue('')
   283→   283→    setFpValue('')
   284→   284→    if (tab === 'barcode') barcodeInputRef.current?.focus()
   285→   285→    else fpInputRef.current?.focus()
   286→   286→  }
   287→   287→
   288→   288→  return (
   289→   289→    <Card className="overflow-hidden">
   290→   290→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'barcode' | 'fingerprint')}>
   291→   291→        <div className="flex flex-col gap-3 border-b bg-muted/30 p-3 sm:flex-row sm:items-center sm:justify-between sm:p-4">
   292→   292→          <TabsList className="grid w-full max-w-md grid-cols-2">
   293→   293→            <TabsTrigger value="barcode" className="gap-2">
   294→   294→              <ScanLine className="h-4 w-4" />
   295→   295→              Student (Barcode)
   296→   296→            </TabsTrigger>
   297→   297→            <TabsTrigger value="fingerprint" className="gap-2">
   298→   298→              <Fingerprint className="h-4 w-4" />
   299→   299→              Teacher (Fingerprint)
   300→   300→            </TabsTrigger>
   301→   301→          </TabsList>
   302→   302→          <Button variant="ghost" size="sm" onClick={onOpenManual} className="gap-2 self-start sm:self-auto">
   303→   303→            <Hand className="h-4 w-4" />
   304→   304→            Manual entry
   305→   305→          </Button>
   306→   306→        </div>
   307→   307→
   308→   308→        <TabsContent value="barcode" className="mt-0 p-4 sm:p-6">
   309→   309→          <div className="grid gap-6 lg:grid-cols-[1.1fr_1fr]">
   310→   310→            {/* Scanner viewport */}
   311→   311→            <div>
   312→   312→              <div className="relative h-56 w-full overflow-hidden rounded-xl border-2 border-primary/30 bg-slate-950 sm:h-64">
   313→   313→                {/* simulated barcode bars background */}
   314→   314→                <div className="absolute inset-0 flex items-center justify-center opacity-40">
   315→   315→                  <div className="flex h-32 w-56 items-center gap-[3px]">
   316→   316→                    {Array.from({ length: 36 }).map((_, i) => (
   317→   317→                      <div
   318→   318→                        key={i}
   319→   319→                        className="bg-slate-100"
   320→   320→                        style={{
   321→   321→                          width: `${((i * 7) % 4) + 1}px`,
   322→   322→                          height: '100%',
   323→   323→                          opacity: i % 2 === 0 ? 1 : 0.2,
   324→   324→                        }}
   325→   325→                      />
   326→   326→                    ))}
   327→   327→                  </div>
   328→   328→                </div>
   329→   329→                {/* scanline laser */}
   330→   330→                <div className="pointer-events-none absolute inset-x-0 top-0 h-1 bg-emerald-400/80 shadow-[0_0_20px_4px_rgba(52,211,153,0.6)] animate-scanline" />
   331→   331→                {/* corner markers */}
   332→   332→                <div className="pointer-events-none absolute top-3 left-3 h-6 w-6 border-t-2 border-l-2 border-emerald-400/70" />

   337→   337→                {flashTick > 0 && (
   338→   338→                  <div
   339→   339→                    key={`flash-${flashTick}`}
   340→   340→                    className="pointer-events-none absolute inset-0 bg-emerald-300/30 animate-in fade-in-0 fade-out-0 duration-500"
   341→   341→                  />

   384→   384→
   385→   385→            {/* Result panel */}
   386→   386→            <ScanResultPanel

   426→   426→                  {teachersLoading

   431→   431→                </Button>
   432→   432→                <div className="relative flex-1">

   446→   446→              <p className="mt-2 text-xs text-muted-foreground">

   731→   731→  }

  1141→  1141→// History View
  1142→  1142→// ════════════════════════════════════════════════════════════════════════════
  1143→  1143→function HistoryView() {
  1144→  1144→  const [date, setDate] = useState<Date>(new Date())
  1145→  1145→  const [personType, setPersonType] = useState('')
  1146→  1146→  const [method, setMethod] = useState('')
  1147→  1147→  const [rows, setRows] = useState<AttendanceRow[]>([])
  1148→  1148→  const [summary, setSummary] = useState({ present: 0, late: 0, absent: 0, total: 0 })
  1149→  1149→  const [loading, setLoading] = useState(true)
  1150→  1150→  const [editTarget, setEditTarget] = useState<AttendanceRow | null>(null)
  1151→  1151→  const [editOpen, setEditOpen] = useState(false)
  1152→  1152→  const [deleteTarget, setDeleteTarget] = useState<AttendanceRow | null>(null)
  1153→  1153→  const [calendarOpen, setCalendarOpen] = useState(false)
  1154→  1154→
  1155→  1155→  const dateStr = useMemo(() => toDateInput(date.toISOString()), [date])
  1156→  1156→
  1157→  1157→  const queryParams = useMemo(() => {
  1158→  1158→    const p = new URLSearchParams()
  1159→  1159→    p.set('date', dateStr)
  1160→  1160→    if (personType) p.set('personType', personType)
  1161→  1161→    if (method) p.set('method', method)
  1162→  1162→    p.set('limit', '200')
  1163→  1163→    return p.toString()
  1164→  1164→  }, [dateStr, personType, method])
  1165→  1165→
  1166→  1166→  // Initial + filter-change fetch. setState happens inside setTimeout + .then
  1167→  1167→  // callbacks only (lint-safe — no synchronous setState in effect body).
  1168→  1168→  useEffect(() => {
  1169→  1169→    let alive = true
  1170→  1170→    const t = setTimeout(() => {
  1171→  1171→      if (!alive) return
  1172→  1172→      setLoading(true)
  1173→  1173→      api<ListResponse>(`/api/attendance?${queryParams}`)
  1174→  1174→        .then((res) => {
  1175→  1175→          if (!alive) return
  1176→  1176→          setRows(res.data)
  1177→  1177→          setSummary(res.summary)
  1178→  1178→        })
  1179→  1179→        .catch((err) => {
  1180→  1180→          if (!alive) return
  1181→  1181→          const msg = err instanceof Error ? err.message : 'Failed to load history'
  1182→  1182→          toast.error(msg)
  1183→  1183→        })
  1184→  1184→        .finally(() => {
  1185→  1185→          if (alive) setLoading(false)
  1186→  1186→        })
  1187→  1187→    }, 0)
  1188→  1188→    return () => {
  1189→  1189→      alive = false
  1190→  1190→      clearTimeout(t)
  1191→  1191→    }
  1192→  1192→  }, [queryParams])
  1193→  1193→
  1194→  1194→  const reload = useCallback(() => {
  1195→  1195→    api<ListResponse>(`/api/attendance?${queryParams}`)
  1196→  1196→      .then((res) => {
  1197→  1197→        setRows(res.data)
  1198→  1198→        setSummary(res.summary)
  1199→  1199→      })
  1200→  1200→      .catch(() => {
  1201→  1201→        /* non-fatal */
  1202→  1202→      })
  1203→  1203→  }, [queryParams])
  1204→  1204→
  1205→  1205→  const handleEdit = (r: AttendanceRow) => {
  1206→  1206→    setEditTarget(r)
  1207→  1207→    setEditOpen(true)
  1208→  1208→  }
  1209→  1209→  const handleDelete = (r: AttendanceRow) => setDeleteTarget(r)
  1210→  1210→
  1211→  1211→  const confirmDelete = async () => {
  1212→  1212→    if (!deleteTarget) return
  1213→  1213→    try {
  1214→  1214→      await api(`/api/attendance/${deleteTarget.id}`, { method: 'DELETE' })
  1215→  1215→      toast.success('Record deleted')
  1216→  1216→      setDeleteTarget(null)
  1217→  1217→      reload()
  1218→  1218→    } catch (err) {
  1219→  1219→      const msg = err instanceof Error ? err.message : 'Delete failed'
  1220→  1220→      toast.error(msg)
  1221→  1221→    }
  1222→  1222→  }
  1223→  1223→
  1224→  1224→  return (
  1225→  1225→    <div className="space-y-4">
  1226→  1226→      {/* Filter bar */}
  1227→  1227→      <div className="flex flex-col gap-3 rounded-xl border bg-card p-3 sm:flex-row sm:items-center sm:gap-2 sm:p-4">
  1228→  1228→        <Popover open={calendarOpen} onOpenChange={setCalendarOpen}>
  1229→  1229→          <PopoverTrigger asChild>
  1230→  1230→            <Button variant="outline" className="justify-start gap-2 sm:w-[200px]">
  1231→  1231→              <CalendarDays className="h-4 w-4" />
  1232→  1232→              {fmtDate(date.toISOString())}
  1233→  1233→            </Button>
  1234→  1234→          </PopoverTrigger>
  1235→  1235→          <PopoverContent className="w-auto p-0" align="start">
  1236→  1236→            <Calendar
  1237→  1237→              mode="single"
  1238→  1238→              selected={date}
  1239→  1239→              onSelect={(d) => {
  1240→  1240→                if (d) {
  1241→  1241→                  setDate(d)
  1242→  1242→                  setCalendarOpen(false)

  1244→  1244→              }}
  1245→  1245→              initialFocus

  1249→  1249→
  1250→  1250→        <Select value={personType || 'ALL'} onValueChange={(v) => setPersonType(v === 'ALL' ? '' : v)}>
  1251→  1251→          <SelectTrigger className="w-full sm:w-[150px]">

  1259→  1259→        </Select>
  1260→  1260→
  1261→  1261→        <Select value={method || 'ALL'} onValueChange={(v) => setMethod(v === 'ALL' ? '' : v)}>
  1262→  1262→          <SelectTrigger className="w-full sm:w-[150px]">

  1265→  1265→          <SelectContent>

  1278→  1278→          onClick={() => {

  1283→  1283→          className="gap-2 self-start sm:ml-auto sm:self-auto"

  1357→  1357→export function AttendanceSection() {

  1359→  1359→  const [manualOpen, setManualOpen] = useState(false)
  1360→  1360→
  1361→  1361→  // Today's log state
  1362→  1362→  const [rows, setRows] = useState<AttendanceRow[]>([])
  1363→  1363→  const [summary, setSummary] = useState({ present: 0, late: 0, absent: 0, total: 0 })

  1381→  1381→  useEffect(() => {
  1382→  1382→    let alive = true
  1383→  1383→    const tick = () => {
  1384→  1384→      api<ListResponse>(`/api/attendance?${todayQuery}`)
  1385→  1385→        .then((res) => {
  1386→  1386→          if (!alive) return
  1387→  1387→          setRows(res.data)
  1388→  1388→          setSummary(res.summary)
  1389→  1389→        })
  1390→  1390→        .catch(() => {
  1391→  1391→          /* non-fatal — keep previous data */
  1392→  1392→        })
  1393→  1393→        .finally(() => {
  1394→  1394→          if (alive) setLoading(false)
  1395→  1395→        })
  1396→  1396→    }
  1397→  1397→    tick()
  1398→  1398→    const id = setInterval(tick, REFRESH_MS)
  1399→  1399→    return () => {
  1400→  1400→      alive = false
  1401→  1401→      clearInterval(id)
  1402→  1402→    }
  1403→  1403→  }, [todayQuery])

  1405→  1405→  // Fetch 7-day trend once for sparklines
  1406→  1406→  useEffect(() => {

  1413→  1413→          teachers: res.trend.map((d) => d.teachers),
  1414→  1414→          late: res.trend.map((d) => d.late),
  1415→  1415→        })
  1416→  1416→      })
  1417→  1417→      .catch(() => {})
  1418→  1418→    return () => {
  1419→  1419→      alive = false
  1420→  1420→    }
  1421→  1421→  }, [])
  1422→  1422→
  1423→  1423→  const reloadToday = useCallback(() => {
  1424→  1424→    api<ListResponse>(`/api/attendance?${todayQuery}`)

  1427→  1427→        setSummary(res.summary)

  1431→  1431→      })

  1439→  1439→    [rows],
  1440→  1440→  )

  1444→  1444→  )

  1447→  1447→    [rows],
  1448→  1448→  )

  1450→  1450→    () => rows.filter((r) => r.checkIn && !r.checkOut).length,
  1451→  1451→    [rows],
  1452→  1452→  )

  1454→  1454→  const handleEdit = (r: AttendanceRow) => {

  1456→  1456→    setEditOpen(true)
  1457→  1457→  }

  1459→  1459→

  1470→  1470→    }
  1471→  1471→  }
  1472→  1472→
  1473→  1473→  return (
  1474→  1474→    <div className="flex flex-col gap-6">
  1475→  1475→      <SectionHeader

  1492→  1492→          footer={
  1493→  1493→            trend.students.length > 0 ? (

  1498→  1498→            ) : undefined
  1499→  1499→          }
  1500→  1500→        />
  1501→  1501→        <StatCard
  1502→  1502→          label="Teachers present"

  1504→  1504→          icon={UserCheck}
  1505→  1505→          accent="purple"
  1506→  1506→          hint="Fingerprint check-in"
  1507→  1507→          footer={
  1508→  1508→            trend.teachers.length > 0 ? (
  1509→  1509→              <div className="flex items-center gap-1.5">
  1510→  1510→                <Sparkline values={trend.teachers} color="#7c3aed" width={64} height={20} />
  1511→  1511→                <span className="text-[10px] text-muted-foreground">7d</span>

  1523→  1523→            trend.late.length > 0 ? (
