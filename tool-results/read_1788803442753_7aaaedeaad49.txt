    93→function toDateInput(iso?: string | null): string {
    94→  if (!iso) return ''
    95→  const d = new Date(iso)
    96→  if (isNaN(d.getTime())) return ''
    97→  const pad = (n: number) => String(n).padStart(2, '0')
    98→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
    99→}
   100→
   101→function toDateTimeInput(iso?: string | null): string {
   102→  if (!iso) return ''
   103→  const d = new Date(iso)
   104→  if (isNaN(d.getTime())) return ''
   105→  const pad = (n: number) => String(n).padStart(2, '0')
   106→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   107→}
   108→
   109→function todayStr(): string {
   110→  const now = new Date()
   111→  const pad = (n: number) => String(n).padStart(2, '0')
   112→  return `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`
   113→}

   281→    setError(null)

   621→// Manual Entry Dialog

   623→interface ManualEntryDialogProps {
   624→  open: boolean
   625→  onOpenChange: (v: boolean) => void
   626→  onSuccess: () => void
   627→}
   628→
   629→function ManualEntryDialog({ open, onOpenChange, onSuccess }: ManualEntryDialogProps) {
   630→  const [personType, setPersonType] = useState<'Student' | 'Teacher'>('Student')
   631→  const [q, setQ] = useState('')
   632→  const [results, setResults] = useState<PersonPick[]>([])
   633→  const [loading, setLoading] = useState(false)
   634→  const [submitting, setSubmitting] = useState(false)
   635→  const searchTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   636→
   637→  // Debounced search
   638→  useEffect(() => {
   639→    if (!open) return
   640→    if (searchTimer.current) clearTimeout(searchTimer.current)
   641→    const query = q.trim()
   642→    if (!query) {
   643→      setResults([])
   644→      return
   645→    }
   646→    let alive = true
   647→    setLoading(true)
   648→    searchTimer.current = setTimeout(() => {
   649→      const endpoint =
   650→        personType === 'Student'
   651→          ? `/api/students?q=${encodeURIComponent(query)}&limit=15`
   652→          : `/api/teachers?q=${encodeURIComponent(query)}&limit=15`
   653→      api<{ data: Array<Record<string, unknown>> }>(endpoint)
   654→        .then((res) => {
   655→          if (!alive) return
   656→          const rows = (res.data || []).map((raw) => {
   657→            if (personType === 'Student') {
   658→              const s = raw as {
   659→                id: string
   660→                studentId: string
   661→                fullName: string
   662→                gender: string
   663→                ageGroup: string | null
   664→                photoUrl: string | null
   665→              }
   666→              return {
   667→                id: s.id,
   668→                ref: s.studentId,
   669→                name: s.fullName,
   670→                type: 'Student' as const,
   671→                photoUrl: s.photoUrl,
   672→                sub: [s.gender, s.ageGroup ? `${s.ageGroup} yrs` : ''].filter(Boolean).join(' · '),
   673→              }
   674→            }
   675→            const t = raw as {
   676→              id: string
   677→              teacherId: string
   678→              fullName: string
   679→              type: string
   680→              photoUrl: string | null
   681→            }
   682→            return {
   683→              id: t.id,
   684→              ref: t.teacherId,
   685→              name: t.fullName,
   686→              type: 'Teacher' as const,
   687→              photoUrl: t.photoUrl,
   688→              sub: t.type,
   689→            }
   690→          })
   691→          setResults(rows)
   692→        })
   693→        .catch(() => {
   694→          if (alive) setResults([])
   695→        })
   696→        .finally(() => {
   697→          if (alive) setLoading(false)
   698→        })
   699→    }, 250)
   700→    return () => {
   701→      alive = false
   702→      if (searchTimer.current) clearTimeout(searchTimer.current)
   703→    }
   704→  }, [q, personType, open])
   705→
   706→  const handlePick = async (p: PersonPick) => {
   707→    setSubmitting(true)
   708→    try {
   709→      const res = await api<ScanResponse>('/api/attendance/scan', {
   710→        method: 'POST',
   711→        body: JSON.stringify({
   712→          method: 'manual',
   713→          value: p.id,
   714→          personType: p.type,
   715→        }),
   716→      })
   717→      const verb = res.action === 'check-in' ? 'Checked in' : 'Checked out'
   718→      toast.success(`${verb}: ${p.name}`, {
   719→        description: `${p.type} · ${p.ref} · Manual entry`,
   720→      })
   721→      onSuccess()
   722→      onOpenChange(false)
   723→      setQ('')
   724→      setResults([])
   725→    } catch (err) {
   726→      const msg = err instanceof Error ? err.message : 'Manual entry failed'
   727→      toast.error(msg)
   728→    } finally {
   729→      setSubmitting(false)
   730→    }
   731→  }
   732→
   733→  return (
   734→    <Dialog open={open} onOpenChange={onOpenChange}>
   735→      <DialogContent className="max-w-lg">
   736→        <DialogHeader>
   737→          <DialogTitle>Manual attendance entry</DialogTitle>
   738→          <DialogDescription>
   739→            Search a student or teacher and mark them present. Use this when the scanner is unavailable.
   740→          </DialogDescription>
   741→        </DialogHeader>
   742→
   743→        <div className="space-y-3">
   744→          <Tabs
   745→            value={personType}
   746→            onValueChange={(v) => {
   747→              setPersonType(v as 'Student' | 'Teacher')
   748→              setQ('')
   749→              setResults([])
   750→            }}
   751→          >
   752→            <TabsList className="grid w-full grid-cols-2">
   753→              <TabsTrigger value="Student" className="gap-2">
   754→                <GraduationCap className="h-4 w-4" />
   755→                Student
   756→              </TabsTrigger>
   757→              <TabsTrigger value="Teacher" className="gap-2">
   758→                <Briefcase className="h-4 w-4" />
   759→                Teacher
   760→              </TabsTrigger>
   761→            </TabsList>
   762→          </Tabs>
   763→
   764→          <div className="relative">
   765→            <Search className="pointer-events-none absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   766→            <Input
   767→              placeholder={`Search ${personType.toLowerCase()} by name, ID or ${personType === 'Student' ? 'barcode' : 'fingerprint'}…`}
   768→              value={q}
   769→              onChange={(e) => setQ(e.target.value)}
   770→              className="pl-9"
   771→              autoFocus
   772→            />
   773→          </div>
   774→
   775→          <div className="max-h-72 overflow-y-auto rounded-lg border">
   776→            {loading ? (
   777→              <div className="space-y-2 p-3">
   778→                {Array.from({ length: 4 }).map((_, i) => (
   779→                  <Skeleton key={i} className="h-12 w-full" />
   780→                ))}
   781→              </div>
   782→            ) : results.length === 0 ? (
   783→              <EmptyState
   784→                title={q ? 'No matches' : 'Start typing to search'}
   785→                description={
   786→                  q
   787→                    ? `No ${personType.toLowerCase()}s match "${q}"`
   788→                    : 'Enter at least 2 characters to search'
   789→                }
   790→                className="border-0 bg-transparent py-8"
   791→              />
   792→            ) : (
   793→              <ul className="divide-y">
   794→                {results.map((p) => (
   795→                  <li key={p.id}>
   796→                    <button
   797→                      type="button"
   798→                      onClick={() => handlePick(p)}
   799→                      disabled={submitting}
   800→                      className="flex w-full items-center gap-3 px-3 py-2.5 text-left hover:bg-muted/60 disabled:opacity-50"
   801→                    >
   802→                      <Avatar className="h-9 w-9">
   803→                        <AvatarFallback className={`text-xs font-bold ${avatarColor(p.name + p.ref)}`}>
   804→                          {initials(p.name)}
   805→                        </AvatarFallback>
   806→                      </Avatar>
   807→                      <div className="min-w-0 flex-1">
   808→                        <p className="truncate text-sm font-semibold">{p.name}</p>
   809→                        <p className="truncate text-xs text-muted-foreground">
   810→                          <span className="font-mono">{p.ref}</span> · {p.sub}
   811→                        </p>
   812→                      </div>
   813→                      <LogIn className="h-4 w-4 text-muted-foreground" />
   814→                    </button>
   815→                  </li>
   816→                ))}
   817→              </ul>
   818→            )}
   819→          </div>
   820→        </div>
   821→
   822→        <DialogFooter>
   823→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   824→            Close
   825→          </Button>
   826→        </DialogFooter>
   827→      </DialogContent>
   828→    </Dialog>
   829→  )
