   102→function toDateInput(iso?: string | null): string {
   103→  if (!iso) return ''
   104→  const d = new Date(iso)
   105→  if (isNaN(d.getTime())) return ''
   106→  return d.toISOString().slice(0, 10)
   107→}
   108→
   109→function csvEscape(v: unknown): string {
   110→  const s = String(v ?? '')
   111→  if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   112→  return s
   113→}

   115→function exportTeachersCsv(rows: TeacherRow[]): void {
   116→  const headers = [
   117→    'Teacher ID',
   118→    'Fingerprint ID',
   119→    'Full Name',
   120→    'Type',
   121→    'Gender',
   122→    'Phone',
   123→    'Email',
   124→    'NIC',
   125→    'Qualification',
   126→    'Specialization',
   127→    'Status',
   128→    'Hire Date',
   129→    'Monthly Rate (LKR)',
   130→    'Classes',
   131→  ]
   132→  const lines = rows.map((r) =>
   133→    [
   134→      r.teacherId,
   135→      r.fingerprintId || '',
   136→      r.fullName,
   137→      r.type,
   138→      r.gender || '',
   139→      r.phone || '',
   140→      r.email || '',
   141→      r.nic || '',
   142→      r.qualification || '',
   143→      r.specialization || '',
   144→      r.status,
   145→      r.hireDate ? toDateInput(r.hireDate) : '',
   146→      String(r.monthlyRate ?? 0),
   147→      String(r._count?.classes ?? 0),
   148→    ]
   149→      .map(csvEscape)
   150→      .join(','),
   151→  )
   152→  const csv = [headers.join(','), ...lines].join('\n')
   153→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   154→  const url = URL.createObjectURL(blob)
   155→  const a = document.createElement('a')
   156→  a.href = url
   157→  a.download = `sanomin-teachers-${new Date().toISOString().slice(0, 10)}.csv`
   158→  document.body.appendChild(a)
   159→  a.click()
   160→  document.body.removeChild(a)
   161→  URL.revokeObjectURL(url)
   162→}
