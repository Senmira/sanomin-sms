   105→   105→function toDateInput(iso?: string | null): string {
   106→   106→  if (!iso) return ''
   107→   107→  const d = new Date(iso)
   108→   108→  if (isNaN(d.getTime())) return ''
   109→   109→  return d.toISOString().slice(0, 10)
   110→   110→}
   111→   111→
   112→   112→function csvEscape(v: unknown): string {
   113→   113→  const s = String(v ?? '')
   114→   114→  if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   115→   115→  return s
   116→   116→}

   118→   118→function exportTeachersCsv(rows: TeacherRow[]): void {
   119→   119→  const headers = [
   120→   120→    'Teacher ID',
   121→   121→    'Fingerprint ID',
   122→   122→    'Full Name',
   123→   123→    'Type',
   124→   124→    'Gender',
   125→   125→    'Phone',
   126→   126→    'Email',
   127→   127→    'NIC',
   128→   128→    'Qualification',
   129→   129→    'Specialization',
   130→   130→    'Status',
   131→   131→    'Hire Date',
   132→   132→    'Monthly Rate (LKR)',
   133→   133→    'Classes',
   134→   134→  ]
   135→   135→  const lines = rows.map((r) =>
   136→   136→    [
   137→   137→      r.teacherId,
   138→   138→      r.fingerprintId || '',
   139→   139→      r.fullName,
   140→   140→      r.type,
   141→   141→      r.gender || '',
   142→   142→      r.phone || '',
   143→   143→      r.email || '',
   144→   144→      r.nic || '',
   145→   145→      r.qualification || '',
   146→   146→      r.specialization || '',
   147→   147→      r.status,
   148→   148→      r.hireDate ? toDateInput(r.hireDate) : '',
   149→   149→      String(r.monthlyRate ?? 0),
   150→   150→      String(r._count?.classes ?? 0),
   151→   151→    ]
   152→   152→      .map(csvEscape)
   153→   153→      .join(','),
   154→   154→  )
   155→   155→  const csv = [headers.join(','), ...lines].join('\n')
   156→   156→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   157→   157→  const url = URL.createObjectURL(blob)
   158→   158→  const a = document.createElement('a')
   159→   159→  a.href = url
   160→   160→  a.download = `sanomin-teachers-${new Date().toISOString().slice(0, 10)}.csv`
   161→   161→  document.body.appendChild(a)
   162→   162→  a.click()
   163→   163→  document.body.removeChild(a)
   164→   164→  URL.revokeObjectURL(url)
   165→   165→}

   875→   875→function AddEditTeacherDialog({
   876→   876→  teacher,
   877→   877→  onClose,
   878→   878→  onSaved,
   879→   879→}: {
   880→   880→  teacher: TeacherRow | null
   881→   881→  onClose: () => void
   882→   882→  onSaved: (msg: string) => void
   883→   883→}) {
   884→   884→  // Lazy initial state — runs once when dialog mounts
   885→   885→  const [form, setForm] = useState<FormState>(() =>
   886→   886→    teacher ? formFromTeacher(teacher) : emptyForm(),
   887→   887→  )
   888→   888→  const [saving, setSaving] = useState(false)
   889→   889→  const [err, setErr] = useState<string | null>(null)
   890→   890→
   891→   891→  const isEdit = teacher !== null
   892→   892→  const isExternal = form.type === 'External'
   893→   893→
   894→   894→  const updateField = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   895→   895→    setForm((f) => ({ ...f, [key]: value }))
   896→   896→  }
   897→   897→
   898→   898→  const handleSubmit = async () => {
   899→   899→    setErr(null)
   900→   900→    if (!form.fullName.trim()) {
   901→   901→      setErr('Full name is required')
   902→   902→      return
   903→   903→    }
   904→   904→
   905→   905→    const monthlyRateNum = Number(form.monthlyRate || 0)
   906→   906→    if (isNaN(monthlyRateNum) || monthlyRateNum < 0) {
   907→   907→      setErr('Monthly rate must be a non-negative number')
   908→   908→      return
   909→   909→    }
   910→   910→
   911→   911→    const payload = {
   912→   912→      fullName: form.fullName.trim(),
   913→   913→      type: form.type,
   914→   914→      gender: form.gender || null,
   915→   915→      phone: form.phone || null,
   916→   916→      email: form.email || null,
   917→   917→      address: form.address || null,
   918→   918→      nic: form.nic || null,
   919→   919→      qualification: form.qualification || null,
   920→   920→      specialization: form.specialization || null,
   921→   921→      status: form.status,
   922→   922→      hireDate: form.hireDate || null,
   923→   923→      monthlyRate: monthlyRateNum,
   924→   924→      fingerprintId: form.fingerprintId || null,
   925→   925→      photoUrl: form.photoUrl || null,
   926→   926→    }
   927→   927→
   928→   928→    setSaving(true)
   929→   929→    try {
   930→   930→      if (isEdit && teacher) {
   931→   931→        const updated = await api<TeacherRow>(
   932→   932→          `/api/teachers/${teacher.id}`,
   933→   933→          { method: 'PUT', body: JSON.stringify(payload) },
   934→   934→        )
   935→   935→        onSaved(`Updated ${updated.fullName}`)
   936→   936→      } else {
   937→   937→        const created = await api<TeacherRow>('/api/teachers', {
   938→   938→          method: 'POST',
   939→   939→          body: JSON.stringify(payload),
   940→   940→        })
   941→   941→        onSaved(`Added ${created.fullName} (${created.teacherId})`)
   942→   942→      }
   943→   943→    } catch (e) {
   944→   944→      const msg = e instanceof Error ? e.message : 'Save failed'
   945→   945→      setErr(msg)
   946→   946→      toast.error(msg)
   947→   947→    } finally {
   948→   948→      setSaving(false)
   949→   949→    }
   950→   950→  }
   951→   951→
   952→   952→  return (
   953→   953→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   954→   954→      <DialogContent className="max-h-[90vh] overflow-y-auto scroll-thin sm:max-w-2xl">

   956→   956→          <DialogTitle>{isEdit ? 'Edit teacher' : 'Add new teacher'}</DialogTitle>

   960→   960→              : 'Add an internal staff member or external tuition teacher. Teacher ID is generated automatically.'}
   961→   961→          </DialogDescription>
   962→   962→        </DialogHeader>
   963→   963→
   964→   964→        {err && (
   965→   965→          <div className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm text-destructive">
   966→   966→            {err}
   967→   967→          </div>
   968→   968→        )}
   969→   969→
   970→   970→        <div className="grid gap-4 py-1">
   971→   971→          {/* Identity */}
   972→   972→          <div className="grid gap-3 sm:grid-cols-2">
   973→   973→            <Field label="Full name *" className="sm:col-span-2">
   974→   974→              <Input
   975→   975→                value={form.fullName}
   976→   976→                onChange={(e) => updateField('fullName', e.target.value)}
   977→   977→                placeholder="e.g. Mrs. Kumari Jayawardena"
   978→   978→              />
   979→   979→            </Field>
   980→   980→            <Field label="Teacher type *">
   981→   981→              <RadioGroup
   982→   982→                value={form.type}
   983→   983→                onValueChange={(v) => updateField('type', v)}
   984→   984→                className="flex gap-4"
   985→   985→              >
   986→   986→                {TEACHER_TYPES.map((t) => (
   987→   987→                  <div key={t} className="flex items-center gap-2">
   988→   988→                    <RadioGroupItem id={`type-${t}`} value={t} />
   989→   989→                    <Label htmlFor={`type-${t}`} className="font-normal">
   990→   990→                      {t}
