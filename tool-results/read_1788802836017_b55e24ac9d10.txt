   836→   836→

   846→   846→  const [studentQuery, setStudentQuery] = useState('')

   946→   946→  )

   948→   948→  // ─── Submit ───────────────────────────────────────────────────────────
   949→   949→  const handleSubmit = async () => {
   950→   950→    if (!form.studentId) {
   951→   951→      toast.error('Please select a student')
   952→   952→      return
   953→   953→    }
   954→   954→    if (!/^\d{4}-\d{2}$/.test(form.month)) {
   955→   955→      toast.error('Month must be in YYYY-MM format')
   956→   956→      return
   957→   957→    }
   958→   958→    const amount = parseFloat(form.amount) || 0
   959→   959→    const paidAmount = parseFloat(form.paidAmount) || 0
   960→   960→    if (amount <= 0) {
   961→   961→      toast.error('Amount must be greater than 0')
   962→   962→      return
   963→   963→    }
   964→   964→    setSaving(true)
   965→   965→    try {
   966→   966→      const payload = {
   967→   967→        studentId: form.studentId,
   968→   968→        programId: form.programId || null,
   969→   969→        month: form.month,

   972→   972→        method: form.method,
   973→   973→        dueDate: form.dueDate || null,
   974→   974→        paidDate: paidAmount > 0 ? form.paidDate || null : null,

   976→   976→      }

   978→   978→        await api(`/api/payments/${payment.id}`, {
   979→   979→          method: 'PUT',
   980→   980→          body: JSON.stringify(payload),
   981→   981→        })
   982→   982→        toast.success('Payment updated')
   983→   983→      } else {
   984→   984→        await api('/api/payments', {
   985→   985→          method: 'POST',
