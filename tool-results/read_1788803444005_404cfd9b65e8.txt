    96→// ─── Form state ─────────────────────────────────────────────────────────────
    97→interface FormState {
    98→  name: string
    99→  programId: string
   100→  teacherId: string
   101→  dayOfWeek: string
   102→  startTime: string
   103→  endTime: string
   104→  room: string
   105→  capacity: string
   106→  fee: string
   107→  active: boolean
   108→  notes: string
   109→}
   110→
   111→const EMPTY_FORM: FormState = {
   112→  name: '',
   113→  programId: '',
   114→  teacherId: '',
   115→  dayOfWeek: 'Mon',
   116→  startTime: '09:00',
   117→  endTime: '10:00',
   118→  room: '',
   119→  capacity: '20',
   120→  fee: '0',
   121→  active: true,
   122→  notes: '',
   123→}

   125→// ─── Main section ───────────────────────────────────────────────────────────
   126→export function ClassesSection() {
   127→  const [rows, setRows] = useState<ClassRow[]>([])
   128→  const [programs, setPrograms] = useState<ProgramRow[]>([])
   129→  const [teachers, setTeachers] = useState<TeacherRow[]>([])
   130→  const [loading, setLoading] = useState(true)
   131→  const [error, setError] = useState<string | null>(null)
   132→
   133→  // Filters
   134→  const [dayFilter, setDayFilter] = useState<string>('all')
   135→  const [programFilter, setProgramFilter] = useState<string>('all')
   136→  const [teacherFilter, setTeacherFilter] = useState<string>('all')
   137→  const [activeFilter, setActiveFilter] = useState<'all' | 'true' | 'false'>('all')
   138→
   139→  // View toggle
   140→  const [view, setView] = useState<'timetable' | 'list'>('timetable')
   141→
   142→  // Dialog state
   143→  const [addEditOpen, setAddEditOpen] = useState(false)
   144→  const [editing, setEditing] = useState<ClassRow | null>(null)
   145→  const [form, setForm] = useState<FormState>(EMPTY_FORM)
   146→  const [saving, setSaving] = useState(false)
   147→  const [deleteTarget, setDeleteTarget] = useState<ClassRow | null>(null)
   148→
   149→  // ─── Initial load: programs + teachers (caches for selects) ──────────────
   150→  useEffect(() => {
   151→    let alive = true
   152→    Promise.all([
   153→      api<ProgramListResponse>('/api/programs'),
   154→      api<TeacherListResponse>('/api/teachers?limit=100'),
   155→    ])
   156→      .then(([p, t]) => {
   157→        if (!alive) return
   158→        setPrograms(p.data)
   159→        setTeachers(t.data)
   160→      })
   161→      .catch(() => {
   162→        // Non-fatal — selects will just be empty
   163→      })
   164→    return () => {
   165→      alive = false
   166→    }
   167→  }, [])
   168→
   169→  // ─── Build query string ──────────────────────────────────────────────────
   170→  const queryParams = useMemo(() => {
   171→    const p = new URLSearchParams()
   172→    p.set('limit', '200')
   173→    if (dayFilter !== 'all') p.set('day', dayFilter)
   174→    if (programFilter !== 'all') p.set('program', programFilter)
   175→    if (teacherFilter !== 'all') p.set('teacherId', teacherFilter)
   176→    if (activeFilter !== 'all') p.set('active', activeFilter)
   177→    return p.toString()
   178→  }, [dayFilter, programFilter, teacherFilter, activeFilter])
   179→
   180→  // ─── Fetch classes (with filters) ────────────────────────────────────────
   181→  useEffect(() => {
   182→    let alive = true
   183→    setLoading(true)
   184→    api<ClassListResponse>(`/api/classes?${queryParams}`)
   185→      .then((res) => {
   186→        if (!alive) return
   187→        setRows(res.data)
   188→        setError(null)
   189→        setLoading(false)
   190→      })
   191→      .catch((err: unknown) => {
   192→        if (!alive) return
   193→        const msg = err instanceof Error ? err.message : 'Failed to load classes'
   194→        setError(msg)
   195→        setLoading(false)
   196→        toast.error(msg)
   197→      })
   198→    return () => {
   199→      alive = false
   200→    }
   201→  }, [queryParams])
   202→
   203→  const reload = useCallback(() => {
   204→    api<ClassListResponse>(`/api/classes?${queryParams}`)
   205→      .then((res) => {
   206→        setRows(res.data)
   207→        setError(null)
   208→      })
   209→      .catch((err: unknown) => {
   210→        const msg = err instanceof Error ? err.message : 'Failed to load classes'
   211→        toast.error(msg)
   212→      })
   213→  }, [queryParams])
   214→
   215→  // ─── Stats ───────────────────────────────────────────────────────────────
   216→  const stats = useMemo(() => {
   217→    let active = 0
   218→    let externalTeacherClasses = 0
   219→    let totalEnrolled = 0
   220→    for (const c of rows) {
   221→      if (c.active) active++
   222→      if (c.teacher?.type === 'External') externalTeacherClasses++
   223→      totalEnrolled += c._count?.enrollments ?? 0
   224→    }
   225→    return {
   226→      total: rows.length,
   227→      active,
   228→      externalTeacherClasses,
   229→      totalEnrolled,
   230→    }
   231→  }, [rows])
   232→
   233→  const hasFilters =
   234→    dayFilter !== 'all' ||
   235→    programFilter !== 'all' ||
   236→    teacherFilter !== 'all' ||
   237→    activeFilter !== 'all'
   238→
   239→  const clearFilters = useCallback(() => {
   240→    setDayFilter('all')
   241→    setProgramFilter('all')
   242→    setTeacherFilter('all')
   243→    setActiveFilter('all')
   244→  }, [])
   245→
   246→  // ─── Dialog handlers ─────────────────────────────────────────────────────
   247→  const openAdd = useCallback(() => {
   248→    setEditing(null)
   249→    setForm(EMPTY_FORM)
   250→    setAddEditOpen(true)
   251→  }, [])
   252→
   253→  const openEdit = useCallback((c: ClassRow) => {
   254→    setEditing(c)
   255→    setForm({
   256→      name: c.name,
   257→      programId: c.program?.id || '',
   258→      teacherId: c.teacher?.id || '',
   259→      dayOfWeek: c.dayOfWeek || 'Mon',
   260→      startTime: c.startTime || '09:00',
   261→      endTime: c.endTime || '10:00',
   262→      room: c.room || '',
   263→      capacity: String(c.capacity ?? 20),
   264→      fee: String(c.fee ?? 0),
   265→      active: c.active,
   266→      notes: c.notes || '',
   267→    })
   268→    setAddEditOpen(true)
   269→  }, [])
   270→
   271→  const closeDialog = useCallback(() => {
   272→    setAddEditOpen(false)
   273→    setEditing(null)
   274→    setForm(EMPTY_FORM)
   275→  }, [])
   276→
   277→  // ─── Save (create/update) ────────────────────────────────────────────────
   278→  const handleSave = useCallback(async () => {
   279→    if (!form.name.trim()) {
   280→      toast.error('Class name is required')
   281→      return
   282→    }
   283→    setSaving(true)
   284→    const payload = {
   285→      name: form.name.trim(),
   286→      programId: form.programId || null,
   287→      teacherId: form.teacherId || null,
   288→      dayOfWeek: form.dayOfWeek,
   289→      startTime: form.startTime,
   290→      endTime: form.endTime,
   291→      room: form.room.trim() || null,
   292→      capacity: Number(form.capacity) || 0,
   293→      fee: Number(form.fee) || 0,
   294→      active: form.active,
   295→      notes: form.notes.trim() || null,
   296→    }
   297→    try {
   298→      if (editing) {
   299→        const updated = await api<ClassRow>(`/api/classes/${editing.id}`, {
   300→          method: 'PUT',
   301→          body: JSON.stringify(payload),
   302→        })
   303→        setRows((prev) => prev.map((c) => (c.id === updated.id ? updated : c)))
   304→        toast.success(`${updated.name} updated`)
   305→      } else {
   306→        const created = await api<ClassRow>('/api/classes', {
   307→          method: 'POST',
   308→          body: JSON.stringify(payload),
   309→        })
   310→        setRows((prev) => [...prev, created])
   311→        toast.success(`${created.name} created`)
   312→      }
   313→      closeDialog()
   314→    } catch (err) {
   315→      const msg = err instanceof Error ? err.message : 'Save failed'
   316→      toast.error(msg)
   317→    } finally {
   318→      setSaving(false)
   319→    }
   320→  }, [form, editing, closeDialog])
   321→
   322→  // ─── Delete ──────────────────────────────────────────────────────────────
   323→  const handleDelete = useCallback(async () => {
   324→    if (!deleteTarget) return
   325→    try {
   326→      await api(`/api/classes/${deleteTarget.id}`, { method: 'DELETE' })
   327→      setRows((prev) => prev.filter((c) => c.id !== deleteTarget.id))
   328→      toast.success(`${deleteTarget.name} deleted`)
   329→      setDeleteTarget(null)
   330→    } catch (err) {
   331→      const msg = err instanceof Error ? err.message : 'Delete failed'
   332→      toast.error(msg)
   333→    }
   334→  }, [deleteTarget])
   335→
   336→  // ─── Filter bar ──────────────────────────────────────────────────────────
   337→  const FilterBar = (
   338→    <Card className="flex flex-wrap items-end gap-3 p-3">
   339→      <div className="flex items-center gap-1.5 text-sm font-medium text-muted-foreground">
   340→        <Filter className="h-4 w-4" />
   341→        Filters
   342→      </div>
   343→
   344→      <div className="grid gap-1">
   345→        <Label htmlFor="filter-day" className="text-xs">
   346→          Day
   347→        </Label>
   348→        <Select value={dayFilter} onValueChange={setDayFilter}>
   349→          <SelectTrigger id="filter-day" className="w-32">
   350→            <SelectValue />
   351→          </SelectTrigger>
   352→          <SelectContent>
   353→            <SelectItem value="all">All days</SelectItem>
   354→            {DAYS.map((d) => (
   355→              <SelectItem key={d} value={d}>
   356→                {d}
   357→              </SelectItem>
   358→            ))}
   359→          </SelectContent>
   360→        </Select>
   361→      </div>
   362→
   363→      <div className="grid gap-1">
   364→        <Label htmlFor="filter-program" className="text-xs">
   365→          Program
   366→        </Label>
   367→        <Select value={programFilter} onValueChange={setProgramFilter}>
   368→          <SelectTrigger id="filter-program" className="w-40">
   369→            <SelectValue />
   370→          </SelectTrigger>
   371→          <SelectContent>
   372→            <SelectItem value="all">All programs</SelectItem>
   373→            {programs.map((p) => (
   374→              <SelectItem key={p.id} value={p.code}>
   375→                {p.name}
   376→              </SelectItem>
   377→            ))}
   378→          </SelectContent>
   379→        </Select>
   380→      </div>
   381→
   382→      <div className="grid gap-1">
   383→        <Label htmlFor="filter-teacher" className="text-xs">
   384→          Teacher
   385→        </Label>
   386→        <Select value={teacherFilter} onValueChange={setTeacherFilter}>
   387→          <SelectTrigger id="filter-teacher" className="w-48">
   388→            <SelectValue />
   389→          </SelectTrigger>
   390→          <SelectContent>
   391→            <SelectItem value="all">All teachers</SelectItem>
   392→            {teachers.map((t) => (
   393→              <SelectItem key={t.id} value={t.id}>
   394→                {t.fullName} · {t.type}
   395→              </SelectItem>
   396→            ))}
   397→          </SelectContent>
   398→        </Select>
   399→      </div>
   400→
   401→      <div className="grid gap-1">
   402→        <Label htmlFor="filter-active" className="text-xs">
   403→          Status
   404→        </Label>
   405→        <Select
   406→          value={activeFilter}
   407→          onValueChange={(v) => setActiveFilter(v as 'all' | 'true' | 'false')}
   408→        >
   409→          <SelectTrigger id="filter-active" className="w-32">
   410→            <SelectValue />
   411→          </SelectTrigger>
   412→          <SelectContent>
   413→            <SelectItem value="all">All</SelectItem>
   414→            <SelectItem value="true">Active</SelectItem>
   415→            <SelectItem value="false">Inactive</SelectItem>
   416→          </SelectContent>
   417→        </Select>
   418→      </div>
   419→
   420→      {hasFilters && (
   421→        <Button variant="ghost" size="sm" onClick={clearFilters} className="gap-1.5">
   422→          <X className="h-3.5 w-3.5" />
   423→          Clear
   424→        </Button>
   425→      )}
   426→    </Card>
   427→  )
   428→
   429→  return (
   430→    <div className="flex flex-col gap-6">
   431→      <SectionHeader
   432→        title="Classes"
   433→        description="Tuition class schedule for external & internal teachers"
   434→        icon={<CalendarDays className="h-5 w-5" />}
   435→        actions={
   436→          <Button onClick={openAdd} size="sm" className="gap-1.5">
   437→            <Plus className="h-4 w-4" />
   438→            Add Class
   439→          </Button>
   440→        }
   441→      />
   442→
   443→      {/* Stats strip */}
   444→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   445→        <StatCard
   446→          label="Total Classes"
   447→          value={stats.total}
   448→          icon={Layers}
   449→          accent="blue"
   450→          hint="All scheduled classes"
   451→        />
   452→        <StatCard
   453→          label="Active Classes"
   454→          value={stats.active}
   455→          icon={CheckCircle2}
   456→          accent="green"
   457→          hint={`${stats.total - stats.active} inactive`}
   458→        />
   459→        <StatCard
   460→          label="External-Teacher Classes"
   461→          value={stats.externalTeacherClasses}
   462→          icon={GraduationCap}
   463→          accent="purple"
   464→          hint="Tuition classes run by external teachers"
   465→        />
   466→        <StatCard
   467→          label="Total Enrolled"
   468→          value={stats.totalEnrolled}
   469→          icon={Users}
   470→          accent="amber"
   471→          hint="Students across all classes"
   472→        />
   473→      </div>
   474→
   475→      {/* Filter bar */}
   476→      {FilterBar}
   477→
   478→      {/* View toggle */}
   479→      <Tabs value={view} onValueChange={(v) => setView(v as 'timetable' | 'list')}>
   480→        <div className="flex w-full items-center justify-between">
   481→          <TabsList>
   482→            <TabsTrigger value="timetable" className="gap-1.5">
   483→              <LayoutGrid className="h-4 w-4" />
   484→              Timetable
   485→            </TabsTrigger>
   486→            <TabsTrigger value="list" className="gap-1.5">
   487→              <ListIcon className="h-4 w-4" />
   488→              List
   489→            </TabsTrigger>
   490→          </TabsList>
   491→          <span className="text-xs text-muted-foreground">
   492→            {rows.length} class{rows.length === 1 ? '' : 'es'}
   493→          </span>
   494→        </div>
   495→
   496→        <TabsContent value="timetable" className="mt-4">
   497→          {loading ? (
   498→            <Skeleton className="h-[60vh] w-full rounded-xl" />
   499→          ) : error ? (
   500→            <EmptyState
   501→              title="Failed to load classes"
   502→              description={error}
   503→              icon={Layers}
   504→              action={
   505→                <Button onClick={reload} size="sm" variant="outline">
   506→                  Retry
   507→                </Button>
   508→              }
   509→            />
   510→          ) : rows.length === 0 ? (
   511→            <EmptyState
   512→              title="No classes scheduled"
   513→              description="Add your first tuition class to start building the weekly timetable."
   514→              icon={BookOpen}
   515→              action={
   516→                <Button onClick={openAdd} size="sm" className="gap-1.5">
   517→                  <Plus className="h-4 w-4" />
   518→                  Add Class
   519→                </Button>
   520→              }
   521→            />
   522→          ) : (
   523→            <TimetableView rows={rows} />
   524→          )}
   525→        </TabsContent>
   526→
   527→        <TabsContent value="list" className="mt-4">
   528→          {loading ? (
   529→            <Skeleton className="h-[60vh] w-full rounded-xl" />
   530→          ) : error ? (
   531→            <EmptyState
   532→              title="Failed to load classes"
   533→              description={error}
   534→              icon={Layers}
   535→              action={
   536→                <Button onClick={reload} size="sm" variant="outline">
   537→                  Retry
   538→                </Button>
   539→              }
   540→            />
   541→          ) : rows.length === 0 ? (
   542→            <EmptyState
   543→              title="No classes found"
   544→              description="Try adjusting filters or add a new class."
   545→              icon={BookOpen}
   546→              action={
   547→                <Button onClick={openAdd} size="sm" className="gap-1.5">
   548→                  <Plus className="h-4 w-4" />
   549→                  Add Class
   550→                </Button>
   551→              }
   552→            />
   553→          ) : (
   554→            <ClassListTable
   555→              rows={rows}
   556→              onEdit={openEdit}
   557→              onDelete={(c) => setDeleteTarget(c)}
   558→            />
   559→          )}
   560→        </TabsContent>
   561→      </Tabs>
   562→
   563→      {/* Add/Edit dialog */}
   564→      <Dialog open={addEditOpen} onOpenChange={(v) => !v && closeDialog()}>
   565→        <DialogContent className="max-w-2xl">
   566→          <DialogHeader>
   567→            <DialogTitle>{editing ? 'Edit Class' : 'Add Class'}</DialogTitle>
   568→            <DialogDescription>
   569→              {editing
   570→                ? 'Update class schedule and details.'
   571→                : 'Schedule a new tuition class for an external or internal teacher.'}
   572→            </DialogDescription>
   573→          </DialogHeader>
   574→
   575→          <div className="grid gap-4 py-2">
   576→            <div className="grid gap-1.5">
   577→              <Label htmlFor="class-name">Class Name</Label>
   578→              <Input
   579→                id="class-name"
   580→                value={form.name}
   581→                onChange={(e) =>
   582→                  setForm((f) => ({ ...f, name: e.target.value }))
   583→                }
   584→                placeholder="e.g. IT Foundations — Grade 1"
   585→              />
   586→            </div>
   587→
   588→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   589→              <div className="grid gap-1.5">
   590→                <Label htmlFor="class-program">Program</Label>
   591→                <Select
   592→                  value={form.programId || 'none'}
   593→                  onValueChange={(v) =>
   594→                    setForm((f) => ({ ...f, programId: v === 'none' ? '' : v }))
   595→                  }
   596→                >
   597→                  <SelectTrigger id="class-program">
   598→                    <SelectValue placeholder="Select program" />
   599→                  </SelectTrigger>
   600→                  <SelectContent>
   601→                    <SelectItem value="none">— No program —</SelectItem>
   602→                    {programs.map((p) => (
   603→                      <SelectItem key={p.id} value={p.id}>
   604→                        <span
   605→                          className="mr-1.5 inline-block h-2.5 w-2.5 rounded-full align-middle"
   606→                          style={{ backgroundColor: p.color }}
   607→                        />
   608→                        {p.name} ({p.code})
   609→                      </SelectItem>
   610→                    ))}
   611→                  </SelectContent>
   612→                </Select>
   613→              </div>
   614→
   615→              <div className="grid gap-1.5">
   616→                <Label htmlFor="class-teacher">Teacher</Label>
   617→                <Select
   618→                  value={form.teacherId || 'none'}
   619→                  onValueChange={(v) =>
   620→                    setForm((f) => ({ ...f, teacherId: v === 'none' ? '' : v }))
   621→                  }
   622→                >
   623→                  <SelectTrigger id="class-teacher">
   624→                    <SelectValue placeholder="Select teacher" />
   625→                  </SelectTrigger>
   626→                  <SelectContent>
   627→                    <SelectItem value="none">— Unassigned —</SelectItem>
   628→                    {teachers.map((t) => (

   789→function parseHour(time: string | null): number | null {
   790→  if (!time) return null
   791→  const m = /^(\d{1,2}):(\d{2})/.exec(time)
   792→  if (!m) return null
   793→  const h = parseInt(m[1], 10)
   794→  return isNaN(h) ? null : h
   795→}

   797→function TimetableView({ rows }: TimetableViewProps) {

   802→    for (const d of DAYS) map[d] = []
   803→    for (const c of rows) {
   804→      if (c.dayOfWeek && map[c.dayOfWeek]) {
   805→        map[c.dayOfWeek].push(c)
   806→      }
   807→    }
   808→    for (const d of DAYS) {
   809→      map[d].sort((a, b) =>
   810→        (a.startTime || '').localeCompare(b.startTime || ''),
   811→      )

   814→  }, [rows])
   815→
   816→  // Compute active hours (hours that have at least one class) for compact mode

   819→    for (const d of DAYS) {
