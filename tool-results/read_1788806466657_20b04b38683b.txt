    91→    91→function lastNMonths(n: number): string[] {
    92→    92→  const out: string[] = []
    93→    93→  const d = new Date()
    94→    94→  for (let i = 0; i < n; i++) {
    95→    95→    const t = new Date(d.getFullYear(), d.getMonth() - i, 1)
    96→    96→    out.push(`${t.getFullYear()}-${String(t.getMonth() + 1).padStart(2, '0')}`)
    97→    97→  }
    98→    98→  return out
    99→    99→}

   140→   140→// ─── Bill line-item helpers ────────────────────────────────────────────────
   141→   141→type BillProgram = { id: string; code: string; name: string; color: string }
   142→   142→
   143→   143→// Distinct programmes on a bill, from its line items (falls back to the
   144→   144→// legacy single-programme column when items are missing).
   145→   145→function billPrograms(p: PaymentRow): BillProgram[] {
   146→   146→  const out: BillProgram[] = []
   147→   147→  for (const it of p.items ?? []) {
   148→   148→    const prog = it.program
   149→   149→    if (prog && !out.some((x) => x.id === prog.id)) out.push(prog)
   150→   150→  }
   151→   151→  if (out.length === 0 && p.program) out.push(p.program)
   152→   152→  return out
   153→   153→}

   162→   162→// One row per bill line item for receipts (falls back to the legacy
   163→   163→// single-programme amount when items are missing).
   164→   164→function billLines(p: PaymentRow): BillLine[] {
   165→   165→  if (p.items && p.items.length > 0) {
   166→   166→    return p.items.map((it, i) => ({
   167→   167→      key: it.programId ?? `item-${i}`,
   168→   168→      description: it.description ?? it.program?.name ?? 'Fee item',
   169→   169→      amount: it.amount,
   170→   170→      color: it.program?.color ?? null,
   171→   171→    }))
   172→   172→  }
   173→   173→  return [
   174→   174→    {
   175→   175→      key: p.programId ?? 'amount',
   176→   176→      description: p.program?.name ?? 'Fee',
   177→   177→      amount: p.amount,
   178→   178→      color: p.program?.color ?? null,
   179→   179→    },
   180→   180→  ]
   181→   181→}

   911→   911→    payment

   935→   935→  const allPrograms = useMemo<ProgramRow[]>(() => {

   937→   937→    for (const p of programs) map.set(p.id, p)

   949→   949→            active: true,
   950→   950→          })
   951→   951→        }
