    84→// ─── Helpers ──────────────────────────────────────────────────────────────
    85→function currentMonth(): string {
    86→  const d = new Date()
    87→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    88→}
    89→
    90→function lastNMonths(n: number): string[] {
    91→  const out: string[] = []
    92→  const d = new Date()
    93→  for (let i = 0; i < n; i++) {
    94→    const t = new Date(d.getFullYear(), d.getMonth() - i, 1)
    95→    out.push(`${t.getFullYear()}-${String(t.getMonth() + 1).padStart(2, '0')}`)
    96→  }
    97→  return out
    98→}
    99→
   100→function monthLabel(m: string): string {
   101→  const [y, mm] = m.split('-').map((n) => parseInt(n, 10))
   102→  const d = new Date(y, mm - 1, 1)
   103→  return d.toLocaleDateString('en-GB', { month: 'short', year: 'numeric' })
   104→}
   105→
   106→function toIsoDate(d: Date): string {
   107→  return d.toISOString().slice(0, 10)
   108→}

   110→function statusBadgeClasses(status: string): string {
   111→  switch (status) {
   112→    case 'Paid':
   113→      return 'border-transparent bg-emerald-500/15 text-emerald-700 dark:text-emerald-300'
   114→    case 'Partial':
   115→      return 'border-transparent bg-amber-500/15 text-amber-700 dark:text-amber-300'
   116→    case 'Overdue':
   117→      return 'border-transparent bg-red-500/15 text-red-700 dark:text-red-300'
   118→    case 'Pending':
   119→    default:
   120→      return 'border-transparent bg-slate-500/15 text-slate-700 dark:text-slate-300'
   121→  }
   122→}
   123→
   124→function methodBadgeClasses(method: string): string {
   125→  switch (method) {
   126→    case 'Cash':
   127→      return 'border-transparent bg-emerald-500/10 text-emerald-700 dark:text-emerald-300'
   128→    case 'Card':
   129→      return 'border-transparent bg-purple-500/10 text-purple-700 dark:text-purple-300'
   130→    case 'Bank':
   131→      return 'border-transparent bg-blue-500/10 text-blue-700 dark:text-blue-300'
   132→    case 'Online':
   133→      return 'border-transparent bg-amber-500/10 text-amber-700 dark:text-amber-300'
   134→    default:
   135→      return 'border-transparent bg-muted text-muted-foreground'
   136→  }
   137→}

   230→    if (statusFilter !== 'all') params.set('status', statusFilter)
   231→    if (programFilter !== 'all') params.set('program', programFilter)
   232→    if (methodFilter !== 'all') params.set('method', methodFilter)

   238→      setError(null)
   239→      api<PaymentListResponse>(url)
   240→        .then((r) => {
   241→          if (!alive) return

   243→          if (tab === 'outstanding') {
   244→            data = data.filter(
   245→              (p) => p.status === 'Partial' || p.status === 'Pending' || p.status === 'Overdue',

   248→            data = data.filter((p) => p.status === 'Paid')

   267→    const t = setTimeout(run, 0)
   268→    return () => {

   270→      clearTimeout(t)
   271→    }

   771→      {editTarget && (

   775→          programs={programs}
   776→          payment={editTarget}

   779→            setEditTarget(null)

   837→function PaymentDialog({

   846→  const [studentQuery, setStudentQuery] = useState('')
   847→  const [selectedStudent, setSelectedStudent] = useState<StudentRow | null>(null)

   851→    payment

   860→          paidDate: payment.paidDate ? toIsoDate(new Date(payment.paidDate)) : '',
   861→          note: payment.note ?? '',

   863→      : emptyForm(month),

   869→  useEffect(() => {

   917→        const progMatch =

   920→        const fee =
   921→          progMatch && programs.find((p) => p.id === progMatch.id)?.monthlyFee

   925→          ...f,
   926→          studentId: s.id,
   927→          programId,

   939→      setForm((f) => ({
   940→        ...f,
   941→        programId,
   942→        amount: prog ? String(prog.monthlyFee) : f.amount,
   943→      }))
   944→    },
   945→    [programs],
   946→  )

   949→  const handleSubmit = async () => {
   950→    if (!form.studentId) {
   951→      toast.error('Please select a student')
   952→      return
   953→    }
   954→    if (!/^\d{4}-\d{2}$/.test(form.month)) {
   955→      toast.error('Month must be in YYYY-MM format')
   956→      return
   957→    }
   958→    const amount = parseFloat(form.amount) || 0
   959→    const paidAmount = parseFloat(form.paidAmount) || 0
   960→    if (amount <= 0) {
   961→      toast.error('Amount must be greater than 0')
   962→      return
   963→    }

   965→    try {
   966→      const payload = {
   967→        studentId: form.studentId,
   968→        programId: form.programId || null,
   969→        month: form.month,
   970→        amount,
   971→        paidAmount,
   972→        method: form.method,
   973→        dueDate: form.dueDate || null,

   975→        note: form.note.trim() || null,

   978→        await api(`/api/payments/${payment.id}`, {
   979→          method: 'PUT',
   980→          body: JSON.stringify(payload),

   983→      } else {
   984→        await api('/api/payments', {
   985→          method: 'POST',
   986→          body: JSON.stringify(payload),
   987→        })
   988→        toast.success('Payment recorded')
   989→      }
   990→      onSaved()
   991→    } catch (e) {
   992→      toast.error(e instanceof Error ? e.message : 'Failed to save payment')
   993→    } finally {
   994→      setSaving(false)
   995→    }
   996→  }

   998→  const balance = Math.max(0, (parseFloat(form.amount) || 0) - (parseFloat(form.paidAmount) || 0))
   999→
  1000→  return (
  1001→    <Dialog open onOpenChange={(v) => !v && onClose()}>
  1002→      <DialogContent className="max-w-2xl">
  1003→        <DialogHeader>
  1004→          <DialogTitle className="flex items-center gap-2">
  1005→            <Wallet className="h-5 w-5 text-primary" />
  1006→            {isEdit ? 'Edit Payment' : 'Record Payment'}
