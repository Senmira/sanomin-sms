    85→// ─── Helpers ──────────────────────────────────────────────────────────────
    86→function currentMonth(): string {
    87→  const d = new Date()
    88→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    89→}
    90→
    91→function lastNMonths(n: number): string[] {
    92→  const out: string[] = []
    93→  const d = new Date()
    94→  for (let i = 0; i < n; i++) {
    95→    const t = new Date(d.getFullYear(), d.getMonth() - i, 1)
    96→    out.push(`${t.getFullYear()}-${String(t.getMonth() + 1).padStart(2, '0')}`)
    97→  }
    98→  return out
    99→}

   101→function monthLabel(m: string): string {
   102→  const [y, mm] = m.split('-').map((n) => parseInt(n, 10))
   103→  const d = new Date(y, mm - 1, 1)
   104→  return d.toLocaleDateString('en-GB', { month: 'short', year: 'numeric' })
   105→}

   140→// ─── Bill line-item helpers ────────────────────────────────────────────────
   141→type BillProgram = { id: string; code: string; name: string; color: string }
   142→
   143→// Distinct programmes on a bill, from its line items (falls back to the
   144→// legacy single-programme column when items are missing).
   145→function billPrograms(p: PaymentRow): BillProgram[] {
   146→  const out: BillProgram[] = []
   147→  for (const it of p.items ?? []) {
   148→    const prog = it.program
   149→    if (prog && !out.some((x) => x.id === prog.id)) out.push(prog)
   150→  }
   151→  if (out.length === 0 && p.program) out.push(p.program)
   152→  return out
   153→}
   154→
   155→interface BillLine {
   156→  key: string
   157→  description: string
   158→  amount: number
   159→  color: string | null
   160→}
   161→
   162→// One row per bill line item for receipts (falls back to the legacy
   163→// single-programme amount when items are missing).
   164→function billLines(p: PaymentRow): BillLine[] {
   165→  if (p.items && p.items.length > 0) {
   166→    return p.items.map((it, i) => ({
   167→      key: it.programId ?? `item-${i}`,
   168→      description: it.description ?? it.program?.name ?? 'Fee item',
   169→      amount: it.amount,
   170→      color: it.program?.color ?? null,
   171→    }))
   172→  }
   173→  return [
   174→    {
   175→      key: p.programId ?? 'amount',
   176→      description: p.program?.name ?? 'Fee',
   177→      amount: p.amount,
   178→      color: p.program?.color ?? null,
   179→    },
   180→  ]
   181→}

   818→      {/* Create / Edit dialogs */}

   887→// ─── Payment create/edit dialog ────────────────────────────────────────────
   888→interface PaymentDialogProps {
   889→  mode: 'create' | 'edit'
   890→  month: string
   891→  programs: ProgramRow[]
   892→  payment?: PaymentRow | null
   893→  onClose: () => void
   894→  onSaved: () => void
   895→}
   896→
   897→function PaymentDialog({
   898→  mode,
   899→  month,
   900→  programs,
   901→  payment,
   902→  onClose,
   903→  onSaved,
   904→}: PaymentDialogProps) {
   905→  const [students, setStudents] = useState<StudentRow[]>([])
   906→  const [studentQuery, setStudentQuery] = useState('')
   907→  const [selectedStudent, setSelectedStudent] = useState<StudentRow | null>(null)
   908→  const [studentOpen, setStudentOpen] = useState(false)
   909→  const [saving, setSaving] = useState(false)
   910→  const [form, setForm] = useState<FormState>(() =>
   911→    payment
   912→      ? {
   913→          studentId: payment.studentId,
   914→          programIds: Array.from(
   915→            new Set(
   916→              [
   917→                ...(payment.items ?? []).map((it) => it.programId),
   918→                payment.programId,
   919→              ].filter((id): id is string => !!id),
   920→            ),
   921→          ),
   922→          month: payment.month,
   923→          paidAmount: String(payment.paidAmount),
   924→          method: payment.method,
   925→          dueDate: payment.dueDate ? toIsoDate(new Date(payment.dueDate)) : '',
   926→          paidDate: payment.paidDate ? toIsoDate(new Date(payment.paidDate)) : '',
   927→          note: payment.note ?? '',
   928→        }
   929→      : emptyForm(month),
   930→  )

   932→  const isEdit = mode === 'edit' && !!payment

   937→    for (const p of programs) map.set(p.id, p)

   939→      for (const it of payment.items ?? []) {
   940→        const prog = it.program
   941→        if (prog && !map.has(prog.id)) {
   942→          map.set(prog.id, {
   943→            id: prog.id,
   944→            code: prog.code,
   945→            name: prog.name,
   946→            description: null,
   947→            color: prog.color,
   948→            monthlyFee: prog.monthlyFee ?? it.amount,
   949→            active: true,
   950→          })
   951→        }
   952→      }
   953→      if (payment.program && !map.has(payment.program.id)) {
   954→        map.set(payment.program.id, {
   955→          id: payment.program.id,
   956→          code: payment.program.code,
   957→          name: payment.program.name,
   958→          description: null,
   959→          color: payment.program.color,
   960→          monthlyFee: 0,
   961→          active: true,
   962→        })
   963→      }
   964→    }
   965→    return Array.from(map.values())
   966→  }, [programs, isEdit, payment])
   967→
   968→  const selectedPrograms = useMemo(
   969→    () => allPrograms.filter((p) => form.programIds.includes(p.id)),
   970→    [allPrograms, form.programIds],
   971→  )
   972→
   973→  // Bill total = Σ selected programme monthly fees (server recomputes the same)
   974→  const totalAmount = useMemo(
   975→    () => selectedPrograms.reduce((sum, p) => sum + (p.monthlyFee || 0), 0),
   976→    [selectedPrograms],
   977→  )
   978→
   979→  // ─── Programme checkbox toggles / quick actions ──────────────────
   980→  const toggleProgram = useCallback((id: string) => {
   981→    setForm((f) => ({
   982→      ...f,
   983→      programIds: f.programIds.includes(id)
   984→        ? f.programIds.filter((x) => x !== id)
   985→        : [...f.programIds, id],
   986→    }))
   987→  }, [])
   988→
   989→  const selectAllPrograms = useCallback(() => {
   990→    setForm((f) => ({ ...f, programIds: allPrograms.map((p) => p.id) }))
   991→  }, [allPrograms])
   992→
   993→  const clearPrograms = useCallback(() => {
   994→    setForm((f) => ({ ...f, programIds: [] }))
   995→  }, [])
   996→
   997→  // ─── Resolve selected student for edit mode ───────────────────────────
   998→  useEffect(() => {
   999→    if (!isEdit || !payment) return
  1000→    let alive = true
  1001→    api<{ data: StudentRow[] }>(
  1002→      `/api/students?q=${encodeURIComponent(payment.student.studentId)}&limit=5`,
  1003→    )
  1004→      .then((r) => {
  1005→        if (!alive) return
  1006→        const match = (r.data || []).find((s) => s.id === payment.studentId)
  1007→        if (match) setSelectedStudent(match)
  1008→      })
  1009→      .catch(() => {
  1010→        /* ignore */
  1011→      })
  1012→    return () => {
  1013→      alive = false
  1014→    }
  1015→  }, [isEdit, payment])
  1016→
  1017→  // ─── Debounced student search ────────────────────────────────────────
  1018→  useEffect(() => {
  1019→    if (studentQuery.trim().length < 2) {
  1020→      setStudents([])

  1022→    }

  1029→        .catch(() => alive && setStudents([]))
  1030→    }, 250)
  1031→    return () => {
  1032→      alive = false
  1033→      clearTimeout(h)
  1034→    }
  1035→  }, [studentQuery])
  1036→
  1037→  // ─── When student changes, auto-check their enrolled programmes ───────
  1038→  const handleSelectStudent = useCallback(
  1039→    (s: StudentRow) => {
  1040→      setSelectedStudent(s)
  1041→      setStudentOpen(false)
  1042→      setStudentQuery('')
  1043→      setForm((f) => {
  1044→        const enrolled = (s.enrollments ?? [])
  1045→          .map((e) => e.program?.id)
  1046→          .filter((id): id is string => !!id)
  1047→          .filter((id) => programs.some((p) => p.id === id))
  1048→        return {
  1049→          ...f,
  1050→          studentId: s.id,
  1051→          programIds:
  1052→            enrolled.length > 0 ? Array.from(new Set(enrolled)) : f.programIds,
  1053→        }
  1054→      })
  1055→    },
  1056→    [programs],

  1058→

  1061→    if (!form.studentId) {
  1062→      toast.error('Please select a student')
  1063→      return
  1064→    }
  1065→    if (!/^\d{4}-\d{2}$/.test(form.month)) {
  1066→      toast.error('Month must be in YYYY-MM format')
  1067→      return
  1068→    }
  1069→    const programIds = Array.from(new Set(form.programIds))
  1070→    if (programIds.length === 0) {
  1071→      toast.error('Select at least one programme for this bill')
  1072→      return
  1073→    }
  1074→    const paidAmount = parseFloat(form.paidAmount) || 0
  1075→    setSaving(true)
  1076→    try {
  1077→      const payload = {
  1078→        studentId: form.studentId,
  1079→        month: form.month,
  1080→        programIds,
  1081→        paidAmount,
  1082→        method: form.method,
  1083→        dueDate: form.dueDate || null,
  1084→        paidDate: paidAmount > 0 ? form.paidDate || null : null,
  1085→        note: form.note.trim() || null,

  1088→        await api(`/api/payments/${payment.id}`, {
  1089→          method: 'PUT',
  1090→          body: JSON.stringify(payload),
  1091→        })
  1092→        toast.success('Payment updated')
  1093→      } else {
  1094→        await api('/api/payments', {
  1095→          method: 'POST',
  1096→          body: JSON.stringify(payload),
  1097→        })
  1098→        toast.success('Payment recorded')
  1099→      }
  1100→      onSaved()
  1101→    } catch (e) {
  1102→      toast.error(e instanceof Error ? e.message : 'Failed to save payment')
  1103→    } finally {
  1104→      setSaving(false)
  1105→    }
  1106→  }
  1107→
  1108→  const balance = Math.max(0, totalAmount - (parseFloat(form.paidAmount) || 0))
  1109→
  1110→  return (
  1111→    <Dialog open onOpenChange={(v) => !v && onClose()}>
  1112→      <DialogContent className="max-w-2xl">
  1113→        <DialogHeader>
  1114→          <DialogTitle className="flex items-center gap-2">
  1115→            <Wallet className="h-5 w-5 text-primary" />
  1116→            {isEdit ? 'Edit Payment' : 'Record Payment'}
  1117→          </DialogTitle>
  1118→          <DialogDescription>
  1119→            {isEdit
  1120→              ? `Update receipt ${payment?.receiptNo ?? '—'} for ${payment?.student.fullName ?? ''}. Changing the selected programmes recalculates the bill total.`
  1121→              : 'One bill per student — tick one or more programmes and the total is computed automatically. Receipt number is auto-generated.'}
  1122→          </DialogDescription>
  1123→        </DialogHeader>
  1124→
  1125→        <div className="grid gap-4 py-2">
  1126→          {/* Student picker */}
  1127→          <div className="flex flex-col gap-1.5">
  1128→            <Label>Student *</Label>
  1129→            {selectedStudent ? (
  1130→              <div className="flex items-center justify-between gap-2 rounded-lg border p-2.5">
  1131→                <div className="flex items-center gap-2.5">

  1137→                  <div>
  1138→                    <p className="text-sm font-medium">{selectedStudent.fullName}</p>
  1139→                    <p className="font-mono text-[11px] text-muted-foreground">
  1140→                      {selectedStudent.studentId}
  1141→                    </p>
  1142→                  </div>
  1143→                </div>
  1144→                <Button
  1145→                  variant="ghost"
  1146→                  size="sm"
  1147→                  onClick={() => {
  1148→                    if (!isEdit) {
  1149→                      setSelectedStudent(null)
  1150→                      setForm((f) => ({ ...f, studentId: '', programIds: [] }))
  1151→                    }
  1152→                  }}
  1153→                  disabled={isEdit}
  1154→                >
  1155→                  {isEdit ? 'Locked' : 'Change'}
  1156→                </Button>
  1157→              </div>
  1158→            ) : (
  1159→              <div className="flex flex-col gap-2">
  1160→                <div className="relative">
  1161→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
  1162→                  <Input
  1163→                    placeholder="Type at least 2 chars to search students…"
  1164→                    value={studentQuery}
  1165→                    onChange={(e) => setStudentQuery(e.target.value)}
  1166→                    onFocus={() => setStudentOpen(true)}
  1167→                    className="pl-9"
