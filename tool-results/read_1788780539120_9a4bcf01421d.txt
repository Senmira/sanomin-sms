   100→function toDateInput(iso?: string | null): string {
   101→  if (!iso) return ''
   102→  const d = new Date(iso)
   103→  if (isNaN(d.getTime())) return ''
   104→  return d.toISOString().slice(0, 10)
   105→}
   106→
   107→function csvEscape(v: unknown): string {
   108→  const s = String(v ?? '')
   109→  if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   110→  return s
   111→}

   113→function exportStudentsCsv(rows: StudentRow[]): void {
   114→  const headers = [
   115→    'Student ID',
   116→    'Barcode',
   117→    'Index No',
   118→    'Full Name',
   119→    'Gender',
   120→    'DOB',
   121→    'Age Group',
   122→    'Religion',
   123→    'Nationality',
   124→    'Programs',
   125→    'Guardian',
   126→    'Guardian Phone',
   127→    'Status',
   128→  ]
   129→  const lines = rows.map((r) => {
   130→    const g = r.guardians[0]
   131→    return [
   132→      r.studentId,
   133→      r.barcode,
   134→      r.indexNo || '',
   135→      r.fullName,
   136→      r.gender,
   137→      r.dob ? toDateInput(r.dob) : '',
   138→      r.ageGroup || '',
   139→      r.religion || '',
   140→      r.nationality || '',
   141→      r.enrollments
   142→        .map((e) => e.program?.code)
   143→        .filter(Boolean)
   144→        .join('|'),
   145→      g?.name || '',
   146→      g?.phone || '',
   147→      r.status,
   148→    ]
   149→      .map(csvEscape)
   150→      .join(',')
   151→  })
   152→  const csv = [headers.join(','), ...lines].join('\n')
   153→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   154→  const url = URL.createObjectURL(blob)
   155→  const a = document.createElement('a')
   156→  a.href = url
   157→  a.download = `sanomin-students-${new Date().toISOString().slice(0, 10)}.csv`
   158→  document.body.appendChild(a)
   159→  a.click()
   160→  document.body.removeChild(a)
   161→  URL.revokeObjectURL(url)
   162→}

   870→function AddEditStudentDialog({
   871→  student,
   872→  onClose,
   873→  onSaved,
   874→}: {
   875→  student: StudentRow | null
   876→  onClose: () => void
   877→  onSaved: (msg: string) => void
   878→}) {
   879→  // Lazy initial state — runs once when dialog mounts
   880→  const [form, setForm] = useState<FormState>(() =>
   881→    student ? formFromStudent(student) : emptyForm(),
   882→  )
   883→  const [guardians, setGuardians] = useState<GuardianEntry[]>(() =>
   884→    student ? guardiansFromStudent(student) : [{ name: '', phone: '', address: '', relationship: 'Guardian', isPrimary: true }],
   885→  )
   886→  const [programCodes, setProgramCodes] = useState<string[]>(() =>
   887→    student
   888→      ? student.enrollments.map((e) => e.program?.code).filter(Boolean) as string[]
   889→      : [],
   890→  )
   891→  const [saving, setSaving] = useState(false)
   892→  const [err, setErr] = useState<string | null>(null)
   893→
   894→  const isEdit = student !== null
   895→
   896→  const updateField = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   897→    setForm((f) => ({ ...f, [key]: value }))
   898→  }
   899→
   900→  const updateGuardian = (idx: number, patch: Partial<GuardianEntry>) => {
   901→    setGuardians((gs) => gs.map((g, i) => (i === idx ? { ...g, ...patch } : g)))
   902→  }
   903→  const addGuardian = () => {
   904→    setGuardians((gs) => [
   905→      ...gs,
   906→      { name: '', phone: '', address: '', relationship: 'Guardian', isPrimary: false },
   907→    ])
   908→  }
   909→  const removeGuardian = (idx: number) => {
   910→    setGuardians((gs) => {
   911→      if (gs.length === 1) return gs
   912→      const next = gs.filter((_, i) => i !== idx)
   913→      const hasPrimary = next.some((g) => g.isPrimary)
   914→      if (!hasPrimary && next.length > 0) {
   915→        // Replace first slot with a new object (no mutation of existing state)
   916→        next[0] = { ...next[0], isPrimary: true }
   917→      }
   918→      return next
   919→    })
   920→  }
   921→  const setPrimaryGuardian = (idx: number) => {
   922→    setGuardians((gs) => gs.map((g, i) => ({ ...g, isPrimary: i === idx })))
   923→  }
   924→
   925→  const toggleProgram = (code: string) => {
   926→    setProgramCodes((ps) =>
   927→      ps.includes(code) ? ps.filter((c) => c !== code) : [...ps, code],
   928→    )
   929→  }
   930→
   931→  const handleSubmit = async () => {
   932→    setErr(null)
   933→    if (!form.fullName.trim()) {
   934→      setErr('Full name is required')
   935→      return
   936→    }
   937→    if (!form.gender) {
   938→      setErr('Gender is required')
   939→      return
   940→    }
   941→
   942→    const payload = {
   943→      fullName: form.fullName.trim(),
   944→      gender: form.gender,
   945→      dob: form.dob || null,
   946→      ageGroup: form.ageGroup || null,
   947→      admissionDate: form.admissionDate || null,
   948→      religion: form.religion || null,
   949→      nationality: form.nationality || null,
   950→      previousSchool: form.previousSchool || null,
   951→      status: form.status,
   952→      medicalNotes: form.medicalNotes || null,
   953→      photoUrl: form.photoUrl || null,
   954→      indexNo: form.indexNo || null,
   955→      guardians: guardians
   956→        .filter((g) => g.name.trim())
   957→        .map((g) => ({
   958→          name: g.name.trim(),
   959→          phone: g.phone.trim() || 'N/A',
   960→          address: g.address || null,
   961→          relationship: g.relationship || 'Guardian',
   962→          isPrimary: g.isPrimary,
   963→        })),
   964→      programCodes,
   965→    }
   966→
   967→    setSaving(true)
   968→    try {
   969→      if (isEdit && student) {
   970→        const updated = await api<StudentRow>(
   971→          `/api/students/${student.id}`,
   972→          { method: 'PUT', body: JSON.stringify(payload) },
   973→        )
   974→        onSaved(`Updated ${updated.fullName}`)
   975→      } else {
   976→        const created = await api<StudentRow>('/api/students', {
   977→          method: 'POST',
   978→          body: JSON.stringify(payload),
   979→        })
   980→        onSaved(`Added ${created.fullName} (${created.studentId})`)
   981→      }
   982→    } catch (e) {
   983→      const msg = e instanceof Error ? e.message : 'Save failed'
   984→      setErr(msg)
   985→      toast.error(msg)
   986→    } finally {
   987→      setSaving(false)
   988→    }
   989→  }
   990→
   991→  return (
   992→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   993→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   994→        <DialogHeader>
   995→          <DialogTitle>{isEdit ? 'Edit student' : 'Add new student'}</DialogTitle>
   996→          <DialogDescription>
   997→            {isEdit
   998→              ? `Update profile for ${student?.fullName} (${student?.studentId}).`
   999→              : 'Create a new student record. Student ID and barcode are generated automatically.'}
  1000→          </DialogDescription>
  1001→        </DialogHeader>
  1002→
  1003→        {err && (
  1004→          <div className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm text-destructive">
  1005→            {err}
  1006→          </div>
  1007→        )}
  1008→
  1009→        <div className="grid gap-4 py-1">
  1010→          {/* Identity */}
  1011→          <div className="grid gap-3 sm:grid-cols-2">
  1012→            <Field label="Full name *" className="sm:col-span-2">
  1013→              <Input
  1014→                value={form.fullName}
  1015→                onChange={(e) => updateField('fullName', e.target.value)}
  1016→                placeholder="e.g. Mahinda Kumar Rishalini"
  1017→              />
  1018→            </Field>
  1019→            <Field label="Gender *">
  1020→              <RadioGroup
  1021→                value={form.gender}
  1022→                onValueChange={(v) => updateField('gender', v)}
  1023→                className="flex gap-4"
  1024→              >
  1025→                {GENDERS.map((g) => (
  1026→                  <div key={g} className="flex items-center gap-2">
  1027→                    <RadioGroupItem id={`gender-${g}`} value={g} />
  1028→                    <Label htmlFor={`gender-${g}`} className="font-normal">
  1029→                      {g}
  1030→                    </Label>
  1031→                  </div>
  1032→                ))}
  1033→              </RadioGroup>
  1034→            </Field>
  1035→            <Field label="Index No.">
  1036→              <Input
  1037→                value={form.indexNo}
  1038→                onChange={(e) => updateField('indexNo', e.target.value)}
  1039→                placeholder="e.g. 200"
  1040→              />
  1041→            </Field>
