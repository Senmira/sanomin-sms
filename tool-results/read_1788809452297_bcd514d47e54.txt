   511→function downloadCsv(filename: string, rows: (string | number)[][]) {
   512→  const csv = rows
   513→    .map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(','))
   514→    .join('\n')
   515→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   516→  const url = URL.createObjectURL(blob)
   517→  const a = document.createElement('a')
   518→  a.href = url
   519→  a.download = filename
   520→  a.click()
   521→  URL.revokeObjectURL(url)
   522→}

   527→function rateColor(rate: number | null, isFuture: boolean): { bg: string; text: string; border: string } {
   528→  if (isFuture) return { bg: 'transparent', text: 'text-muted-foreground/40', border: 'border-border/40' }
   529→  if (rate === null) return { bg: 'bg-muted/20', text: 'text-muted-foreground/50', border: 'border-border/50' }
   530→  if (rate >= 90) return { bg: 'bg-emerald-500/25 dark:bg-emerald-500/30', text: 'text-emerald-700 dark:text-emerald-300', border: 'border-emerald-500/40' }
   531→  if (rate >= 75) return { bg: 'bg-emerald-500/15 dark:bg-emerald-500/20', text: 'text-emerald-600 dark:text-emerald-400', border: 'border-emerald-500/30' }
   532→  if (rate >= 50) return { bg: 'bg-amber-500/15 dark:bg-amber-500/20', text: 'text-amber-600 dark:text-amber-400', border: 'border-amber-500/30' }
   533→  if (rate > 0) return { bg: 'bg-red-500/15 dark:bg-red-500/20', text: 'text-red-600 dark:text-red-400', border: 'border-red-500/30' }
   534→  return { bg: 'bg-muted/30', text: 'text-muted-foreground', border: 'border-border' }
   535→}

   537→function HeatmapPanel() {
   538→  const now = new Date()
   539→  const [month, setMonth] = useState(`${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`)
   540→  const [data, setData] = useState<HeatmapReport | null>(null)
   541→  const [loading, setLoading] = useState(true)
   542→  const [hovered, setHovered] = useState<HeatmapCell | null>(null)
   543→
   544→  useEffect(() => {
   545→    let alive = true
   546→    Promise.resolve().then(() => alive && setLoading(true))
   547→    api<HeatmapReport>(`/api/reports?type=heatmap&month=${month}`)
   548→      .then((d) => alive && setData(d))
   549→      .catch(() => alive && setData(null))
   550→      .finally(() => alive && setLoading(false))
   551→    return () => {
   552→      alive = false
   553→    }
   554→  }, [month])
   555→
   556→  const goPrev = () => {
   557→    const [y, m] = month.split('-').map(Number)
   558→    const d = new Date(y, m - 1, 1)
   559→    d.setMonth(d.getMonth() - 1)
   560→    setMonth(`${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`)
   561→  }

   563→    const [y, m] = month.split('-').map(Number)
   564→    const d = new Date(y, m - 1, 1)
   565→    d.setMonth(d.getMonth() + 1)
   566→    setMonth(`${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`)
   567→  }
   568→  const goToday = () => setMonth(`${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`)
   569→
   570→  if (loading || !data) {
   571→    return (
   572→      <div className="grid gap-4">

   575→      </div>
   576→    )
   577→  }
   578→
   579→  const s = data.summary
   580→  const isCurrentMonth = month === `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`
   581→
   582→  return (
